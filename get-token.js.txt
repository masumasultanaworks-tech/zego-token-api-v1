const crypto = require("crypto");

function generateToken04(appId, userId, secret, effectiveTimeInSeconds, payload) {
    if (!appId || !userId || !secret) {
        throw new Error("AppID, UserID and AppSecret are required.");
    }
    const createTime = Math.floor(Date.now() / 1000);
    const expireTime = createTime + Number(effectiveTimeInSeconds);
    const nonce = Math.floor(Math.random() * 2147483647);

    const tokenInfo = {
        app_id: Number(appId),
        user_id: String(userId),
        nonce: nonce,
        ctime: createTime,
        expire: expireTime,
        payload: payload || ""
    };

    const plainText = JSON.stringify(tokenInfo);
    const iv = crypto.randomBytes(16);

    let keyBuffer;
    if (typeof secret === "string" && secret.length === 64) {
        keyBuffer = Buffer.from(secret, "hex").subarray(0, 16);
    } else {
        keyBuffer = Buffer.from(secret, "utf8").subarray(0, 16);
    }

    const cipher = crypto.createCipheriv("aes-128-cbc", keyBuffer, iv);
    let encrypted = cipher.update(plainText, "utf8", "base64");
    encrypted += cipher.final("base64");

    const ivBase64 = iv.toString("base64");
    return `04${ivBase64}${encrypted}`;
}

module.exports = async (req, res) => {
    // CORS অনুমোদন দেওয়া হচ্ছে যেন Flutter অ্যাপ থেকে সহজে কল করা যায়
    res.setHeader("Access-Control-Allow-Origin", "*");
    res.setHeader("Access-Control-Allow-Methods", "POST, OPTIONS");
    res.setHeader("Access-Control-Allow-Headers", "Content-Type");

    if (req.method === "OPTIONS") {
        return res.status(200).end();
    }

    if (req.method !== "POST") {
        return res.status(405).json({ error: "অনুরোধ অবশ্যই POST হতে হবে।" });
    }

    try {
        const { userId } = req.body || {};

        if (!userId) {
            return res.status(400).json({ error: "userId দেওয়া আবশ্যক।" });
        }

        // Vercel Environment Variables থেকে সিক্রেট নেওয়া হচ্ছে
        const appId = process.env.ZEGO_APP_ID;
        const appSecret = process.env.ZEGO_APP_SECRET;

        if (!appId || !appSecret) {
            return res.status(500).json({ error: "সার্ভারে কনফিগারেশন এরর রয়েছে।" });
        }

        const effectiveTime = 3600; // ১ ঘণ্টা
        const token = generateToken04(appId, userId, appSecret, effectiveTime, "");

        return res.status(200).json({ token });
    } catch (error) {
        console.error("Token Generation Error:", error.message);
        return res.status(500).json({ error: "টোকেন জেনারেট করা সম্ভব হয়নি।" });
    }
};
